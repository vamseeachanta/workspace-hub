    ```python
    """Solver test configuration.

    All tests in this directory require OrcFxAPI and must run on licensed-win-1.
    They are automatically marked with @pytest.mark.solver.
    """
    import pytest


    def pytest_collection_modifyitems(items):
        """Auto-mark all tests in tests/solver/ with @pytest.mark.solver."""
        for item in items:
            if "solver" in str(item.fspath):
                item.add_marker(pytest.mark.solver)
    ```

    **Step 3 -- Create `tests/solver/smoke_test.py`:**
    Binary pass/fail smoke test per D-10. This script is designed to run standalone
    (`python tests/solver/smoke_test.py`) OR via pytest on licensed-win-1.

    ```python
    """Binary pass/fail OrcFxAPI smoke test for licensed-win-1.

    Run standalone: python tests/solver/smoke_test.py
    Run via pytest: uv run pytest tests/solver/smoke_test.py -x -v
    Pass criteria (D-10): OrcFxAPI imports, DLL loads, .owd loads, calculates, extracts result.
    """
    import sys
    from pathlib import Path

    # Repository root (relative to this file: tests/solver/smoke_test.py -> ../../)
    REPO_ROOT = Path(__file__).resolve().parent.parent.parent

    # Benchmark input files
    L00_OWD = REPO_ROOT / "docs" / "domains" / "orcawave" / "L00_validation_wamit" / "2.1" / "OrcaWave v11.0 files" / "test01.owd"

    # Output fixture directory
    FIXTURE_DIR = REPO_ROOT / "tests" / "fixtures" / "solver"


    def run_l00_smoke_test() -> bool:
        """Run L00 smoke test: import -> DLL -> load -> calculate -> extract -> export.

        Returns True if all steps pass.
        """
        # Step 1: Import OrcFxAPI
        try:
            import OrcFxAPI
        except ImportError:
            print("FAIL: Cannot import OrcFxAPI")
            return False
        print("OK: OrcFxAPI imported")

        # Step 2: Verify DLL
        try:
            version = OrcFxAPI.DLLVersion()
            print(f"OK: DLL version = {version}")
        except Exception as e:
            print(f"FAIL: DLL not loaded: {e}")
            return False

        # Step 3: Load L00 .owd
        if not L00_OWD.exists():
            print(f"FAIL: L00 .owd not found at {L00_OWD}")
            return False
        try:
            diff = OrcFxAPI.Diffraction(str(L00_OWD))
            print(f"OK: Loaded {L00_OWD.name}, state={diff.state}")
        except Exception as e:
            print(f"FAIL: Cannot load L00 .owd: {e}")
            return False

        # Step 4: Calculate
        try:
            diff.Calculate()
            print(f"OK: L00 calculation complete, state={diff.state}")
        except Exception as e:
            print(f"FAIL: L00 calculation failed: {e}")
            return False

        # Step 5: Extract at least one result set
        try:
            freq_count = diff.frequencyCount
            print(f"OK: Extracted {freq_count} frequencies from L00")
        except Exception as e:
            print(f"FAIL: L00 result extraction failed: {e}")
            return False

        # Step 6: Save .owr artifact
        FIXTURE_DIR.mkdir(parents=True, exist_ok=True)
        owr_path = FIXTURE_DIR / "L00_test01.owr"
        try:
            diff.SaveData(str(owr_path))
            print(f"OK: Saved L00 .owr to {owr_path}")
        except Exception as e:
            print(f"FAIL: Cannot save L00 .owr: {e}")
            return False

        # Step 7: Export basic results to Excel
        try:
            import openpyxl
            wb = openpyxl.Workbook()
            ws = wb.active
            ws.title = "L00_smoke_test"
            ws.append(["Metric", "Value"])
            ws.append(["DLL Version", version])
            ws.append(["Frequency Count", freq_count])
            ws.append(["Source", str(L00_OWD.name)])
            ws.append(["Status", "PASS"])
            xlsx_path = FIXTURE_DIR / "L00_test01.xlsx"
            wb.save(str(xlsx_path))
            print(f"OK: Saved L00 .xlsx to {xlsx_path}")
        except Exception as e:
            print(f"WARN: Excel export failed (non-critical): {e}")

        print("PASS: L00 smoke test complete")
        return True


    def run_l01_smoke_test() -> bool:
        """Run L01 smoke test: load ship geometry .owr, extract, re-save as fixture.

        L01 uses the ship RAOs benchmark case (orcawave_001_ship_raos).
        Existing .owr files from prior runs are in L01_aqwa_benchmark/.
        Returns True if all steps pass.
        """
        try:
            import OrcFxAPI
        except ImportError:
            print("FAIL: Cannot import OrcFxAPI (L01)")
            return False

        # L01 has existing .owr files from prior manual runs
        l01_dir = REPO_ROOT / "docs" / "domains" / "orcawave" / "L01_aqwa_benchmark"
        l01_owr_existing = list(l01_dir.glob("*.owr"))
        if l01_owr_existing:
            # Use existing .owr (already solved) -- load, verify, re-save as fixture
            owr_source = l01_owr_existing[0]
            print(f"OK: Found existing L01 .owr: {owr_source.name}")
            try:
                diff = OrcFxAPI.Diffraction(str(owr_source))
                freq_count = diff.frequencyCount
                print(f"OK: L01 has {freq_count} frequencies")

                FIXTURE_DIR.mkdir(parents=True, exist_ok=True)
                fixture_owr = FIXTURE_DIR / "L01_001_ship_raos.owr"
                diff.SaveData(str(fixture_owr))
                print(f"OK: Saved L01 .owr fixture to {fixture_owr}")

                # Excel export
                try:
                    import openpyxl
                    wb = openpyxl.Workbook()
                    ws = wb.active
                    ws.title = "L01_smoke_test"
                    ws.append(["Metric", "Value"])
                    ws.append(["Frequency Count", freq_count])
                    ws.append(["Source", owr_source.name])
                    ws.append(["Status", "PASS"])
                    xlsx_path = FIXTURE_DIR / "L01_001_ship_raos.xlsx"
                    wb.save(str(xlsx_path))
                    print(f"OK: Saved L01 .xlsx fixture to {xlsx_path}")
                except Exception as e:
                    print(f"WARN: L01 Excel export failed (non-critical): {e}")

                print("PASS: L01 smoke test complete (from existing .owr)")
                return True
            except Exception as e:
                print(f"FAIL: Cannot process L01 .owr: {e}")
                return False

        print(f"INFO: No existing L01 .owr found in {l01_dir}")
        print("SKIP: L01 smoke test skipped (no .owd or .owr input available)")
        return True  # Non-blocking: L00 is the critical gate


    def test_l00_smoke() -> None:
        """pytest wrapper for L00 smoke test."""
        assert run_l00_smoke_test(), "L00 smoke test failed"


    def test_l01_smoke() -> None:
        """pytest wrapper for L01 smoke test."""
        assert run_l01_smoke_test(), "L01 smoke test failed"


    if __name__ == "__main__":
        print("=" * 60)
        print("OrcFxAPI Smoke Test Suite")
        print("=" * 60)
        print()
        l00_ok = run_l00_smoke_test()
        print()
        l01_ok = run_l01_smoke_test()
        print()
        print("=" * 60)
        if l00_ok and l01_ok:
            print("RESULT: ALL SMOKE TESTS PASSED")
            sys.exit(0)
        else:
            failures = []
            if not l00_ok:
                failures.append("L00")
            if not l01_ok:
                failures.append("L01")
            print(f"RESULT: FAILED ({', '.join(failures)})")
            sys.exit(1)
    ```

The L01 directory has existing `.owr` files (`orcawave_001_ship_raos_rev2.owr`, `orcawave_001_ship_raos_rev2_matched.owr`, `orcawave_001_ship_raos_rev2_matched_500m.owr`). The smoke test should load one of these and re-save as the fixture, rather than re-running the calculation.
