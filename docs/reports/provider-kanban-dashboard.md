
```sh
uv run --no-project python scripts/ai/approve-provider-plan.py \
  --issue <N> --mode dry-run    # validate planned transaction
uv run --no-project python scripts/ai/approve-provider-plan.py \
  --issue <N> --mode real \
  --user-identity <email> --approval-source 'cli' --confirmation-token <tty>
```
