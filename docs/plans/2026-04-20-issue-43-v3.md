```
# Parse logs into SessionRecord list using the log-format contract.
function parse_session_logs(start_date, end_date, signal_roots):
    sessions = []
    for file in walk([".claude/state/session-signals/", "state/session-signals/"]):
        date_in_name = extract_date(file.name)
        if date_in_name not in [start_date..end_date]: continue
        provider = infer_provider(file, event_attr="session_id" | path_pattern)
        events = parse_jsonl_with_contract(file, LOG_FORMAT_CONTRACT)
        sessions.append(SessionRecord(provider, events, file))
    return sessions
```

```
# Scorer: verified event taxonomy only. 3 derivable dims + 3 insufficient_corpus placeholders.
function score_provider(sessions, provider):
    if len(sessions) == 0:
        return {"status": "insufficient_data", "provider": provider,
                "reason": "zero sessions in 2026-03-08..2026-03-15 window"}
    total = len(sessions)

    lifecycle = mean(
        len({e.stage for e in s.events if e.event == "stage_exit"} & set(range(1, 21))) / 20
        for s in sessions)

    review_evts = [e for s in sessions for e in s.events
                   if e.event == "stage_exit" and e.stage in {5, 7, 17}]
    user_review_adherence = (sum(1 for e in review_evts if e.next_gap_seconds >= 60)
                             / len(review_evts)) if review_evts else None

    log_completeness = sum(1 for s in sessions
                           if any(e.event == "session_tool_summary" for e in s.events)) / total

    insufficient = {
        "gate_first_pass":       {"status": "insufficient_corpus",
                                  "missing_events": ["verify_gate_evidence_result"]},
        "tdd_faithfulness":      {"status": "insufficient_corpus",
                                  "missing_events": ["test_write_event", "impl_write_event"]},
        "tool_skill_invocation": {"status": "insufficient_corpus",
                                  "missing_events": ["skill_invoke"]},
    }
    return {"provider": provider, "n_sessions": total,
            "lifecycle": lifecycle,
            "user_review_adherence": user_review_adherence,
            "log_completeness": log_completeness,
            **insufficient}
```

```
# Hard-gate violations: explicit list, not just a ratio.
function find_hard_gate_violations(sessions) -> list[Violation]:
    violations = []
    for s in sessions:
        for e in s.events:
            if e.stage in {5, 7, 17} and e.type == "gate_bypass":
                violations.append(Violation(
                    session_id=s.id, stage=e.stage,
                    evidence_path=s.log_path, timestamp=e.ts,
                    provider=s.provider, excerpt=e.raw_line))
    return violations
```

```
# Emitter: latest-wins replace-by-key. Mid-day reruns overwrite stale records.
function emit_signals(scores, date, dry_run=True):
    path = f".claude/state/session-signals/43-compliance-{date}.jsonl"

    new_records = []
    for p, s in per_provider_scores(scores):
        new_records.append({"event":"gate_compliance_score", "provider":p,
                            "score":s, "date":date, "run_ts": now_iso()})
    for p, c in per_provider_lifecycle(scores):
        new_records.append({"event":"provider_capability", "provider":p,
                            "lifecycle_complete":c, "date":date, "run_ts": now_iso()})

    keys_to_replace = {(r["event"], r["provider"], r["date"]) for r in new_records}
    existing = read_jsonl_if_exists(path)
    retained = [r for r in existing
                if (r["event"], r["provider"], r["date"]) not in keys_to_replace]

    if dry_run:
        return {"status": "dry_run", "would_replace": len(new_records),
                "retained": len(retained), "path": path}

    atomic_write_jsonl(path, retained + new_records)      # tmp → fsync → rename
    return {"status": "replaced", "n_new": len(new_records),
            "n_retained": len(retained), "path": path}
```

```
function emit_routing_rules(scores, date, out_path="config/agents/routing-rules.yaml"):
    header = (
        "# GENERATED — DO NOT EDIT\n"

        "#   3. THIS FILE (advisory snapshot from latest compliance run)\n"
        f"# Last generated: {now_iso()} from .claude/state/session-signals/43-compliance-{date}.jsonl\n"
    )
    write_yaml_with_header(out_path, header, {"assessment": scores, "version": "1.0.0"})
```
